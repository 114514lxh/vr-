local function naturalSway(t, seed)
    return math.sin(t * 0.63 + seed) * 0.55
         + math.sin(t * 1.19 + seed * 2.71) * 0.28
         + math.sin(t * 1.87 + seed * 5.13) * 0.17
end
local function slowDrift(t, seed)
    return math.sin(t * 0.11 + seed) * 0.5
         + math.sin(t * 0.23 + seed * 2.31) * 0.3
         + math.sin(t * 0.43 + seed * 3.77) * 0.2
end
local function breathDrift(t, seed)
    return math.sin(t * 0.07 + seed) * 0.6
         + math.sin(t * 0.13 + seed * 1.87) * 0.4
end

local function sampleGesture(g, t)
    local kfs = g.kfs
    if t <= kfs[1][1] then return kfs[1][2], kfs[1][3] end
    if t >= kfs[#kfs][1] then return kfs[#kfs][2], kfs[#kfs][3] end
    for i = 1, #kfs - 1 do
        local a, b = kfs[i], kfs[i+1]
        if t >= a[1] and t <= b[1] then
            local f = (t - a[1]) / (b[1] - a[1])
            f = f * f * (3 - 2 * f)
            return a[2]:Lerp(b[2], f), a[3]:Lerp(b[3], f)
        end
    end
    return Vector3.zero, CFrame.new()
end

        local function applyWheelDrag(drag, pos)
            if drag.kind == "wheel" then
                local wf = getWheelFrame(drag.side)
                local kn = getWheelKnob(drag.side)
                if not wf or not kn then return end
                local ap, as = wf.AbsolutePosition, wf.AbsoluteSize
                local cx, cy = ap.X + as.X * 0.5, ap.Y + as.Y * 0.5
                local r = as.X * 0.5
                local maxR = r * 0.75
                local dx, dy = pos.X - cx, pos.Y - cy
                local dist = math.sqrt(dx * dx + dy * dy)
                if dist > maxR then dx, dy = dx / dist * maxR, dy / dist * maxR end
                local nx, ny = dx / maxR, dy / maxR
                if math.abs(nx) < 0.10 then nx = 0 end
                if math.abs(ny) < 0.10 then ny = 0 end
                kn.Position = UDim2.new(0.5, nx * maxR, 0.5, ny * maxR)

                local mag = magRaw * magRaw

                local off = wheelOffset[drag.side]
                off.x =  nx * WHEEL_MAX_XY * mag
                off.z = -ny * WHEEL_MAX_FWD * mag

                local rot = wheelRot[drag.side]
                rot.yaw   =  nx * WHEEL_ROT_MAX * mag
                rot.pitch = -ny * WHEEL_ROT_MAX * mag
                rot.roll  = -nx * WHEEL_ROLL_MAX * mag * 0.6

                local off = wheelOffset[drag.side]

            end

    RunService:BindToRenderStep("NoVR_Control", Enum.RenderPriority.Camera.Value + 1, function(dt)
        local t = os.clock()
        frameCounter = frameCounter + 1

        if recordMode then
            if t - recordStartT > RECORD_MAX_DURATION then
                recordMode = false
                print(("[NoVR] 录制超时，共 %d 帧"):format(maxRecordFrame))
                if maxRecordFrame > 0 and refreshLibraryListRef then
                    recordCounter = recordCounter + 1
                    table.insert(recordLibrary, {
                        name = "录制 " .. recordCounter,
                        frames = recordFrames,
                        maxFrame = maxRecordFrame,
                        duration = maxRecordFrame / 60,
                    })
                    if #recordLibrary > RECORD_LIBRARY_MAX then
                        table.remove(recordLibrary, 1)
                    end
                    refreshLibraryListRef()
                end
            end
        end
        if playbackMode then
            playbackFrame = playbackFrame + 1
            if playbackFrame > maxRecordFrame then
                playbackMode = false
                print("[NoVR] 回放结束")
            end
        end

        if S.look and not UIS.TouchEnabled then
            local d = UIS:GetMouseDelta()
            local comp = math.clamp(dt * 60, 0.25, 3)
            targetYaw   = targetYaw   - d.X * S.sens * comp
            targetPitch = math.clamp(targetPitch - d.Y * S.sensPitch * comp, -1.55, 1.55)
            UIS.MouseBehavior = Enum.MouseBehavior.LockCenter
        end

        if headGyroEnabled and gyroNow then
            if not headGyroBase then
                headGyroBase = gyroNow
                headGyroLastYaw = 0
                headGyroLastPitch = 0
            end
            local rel = headGyroBase:Inverse() * gyroNow
            local rx, ry, rz = rel:ToEulerAnglesYXZ()

            local dPitch = rx - headGyroLastPitch
            if dPitch > math.pi then dPitch = dPitch - math.pi * 2
            elseif dPitch < -math.pi then dPitch = dPitch + math.pi * 2 end

            local dYaw = ry - headGyroLastYaw
            if dYaw > math.pi then dYaw = dYaw - math.pi * 2
            elseif dYaw < -math.pi then dYaw = dYaw + math.pi * 2 end

            if math.abs(dYaw) < 0.5 then
                targetYaw = targetYaw + dYaw * S.headGyroSens
            end
            if math.abs(dPitch) < 0.5 then
                targetPitch = math.clamp(targetPitch - dPitch * S.headGyroSens, -1.55, 1.55)
            end

            headGyroLastYaw = ry
            headGyroLastPitch = rx
        else
            headGyroBase = nil
            headGyroLastYaw = 0
            headGyroLastPitch = 0
        end

        if wheelHandEnabled and WHEEL_AUTO_RETURN then
            local dragging = _G.__NoVR_wheelDragging and _G.__NoVR_wheelDragging()
            if not dragging then
                local retK = 1 - math.exp(-WHEEL_RETURN_SPEED * dt)
                for _, side in ipairs({"L", "R"}) do
                    local off = wheelOffset[side]
                    off.x = off.x * (1 - retK)
                    off.y = off.y * (1 - retK)
                    off.z = off.z * (1 - retK)
                    local rot = wheelRot[side]
                    rot.yaw   = rot.yaw   * (1 - retK)
                    rot.pitch = rot.pitch * (1 - retK)
                    rot.roll  = rot.roll  * (1 - retK)
                end
            end
        end

        do
            local wKPos = 1 - math.exp(-6 * dt)
            local wKRot = 1 - math.exp(-3.5 * dt)
            for _, side in ipairs({"L", "R"}) do
                local off = wheelOffset[side]
                local oS = wheelOffsetSmooth[side]
                oS.x = oS.x + (off.x - oS.x) * wKPos
                oS.y = oS.y + (off.y - oS.y) * wKPos
                oS.z = oS.z + (off.z - oS.z) * wKPos

                local rot = wheelRot[side]
                local rS = wheelRotSmooth[side]
                rS.yaw   = rS.yaw   + (rot.yaw   - rS.yaw)   * wKRot
                rS.pitch = rS.pitch + (rot.pitch - rS.pitch) * wKRot
                rS.roll  = rS.roll  + (rot.roll  - rS.roll)  * wKRot

                local wOff = wheelOffsetSmooth[side]
                local wVec = Vector3.new(wOff.x, wOff.y, -wOff.z)
                local targetPitch = wVec.z * 0.55 + wVec.y * 0.25
                local targetYaw   = -wVec.x * 0.50
                local targetRoll  = -wVec.x * 0.35
                local tS = handTiltSmooth[side]
                local tiltK = 1 - math.exp(-8 * dt)
                tS.pitch = tS.pitch + (targetPitch - tS.pitch) * tiltK
                tS.yaw   = tS.yaw   + (targetYaw   - tS.yaw)   * tiltK
                tS.roll  = tS.roll  + (targetRoll  - tS.roll)  * tiltK
            end

            if wheelHandEnabled then
                local dragging = _G.__NoVR_wheelDragging and _G.__NoVR_wheelDragging()
                if not dragging and _G.__NoVR_wheelUi then
                    local ui = _G.__NoVR_wheelUi
                    local maxR = 140 * 0.5 * 0.75
                    local halfSlider = 70 - 14
                    for _, side in ipairs({"L", "R"}) do
                        local oS = wheelOffsetSmooth[side]
                        local kn = (side == "L") and ui.wheelLKnob or ui.wheelRKnob
                        if kn then
                            local nx = math.clamp(oS.x / WHEEL_MAX_XY, -1, 1)
                            local ny = math.clamp(-oS.z / WHEEL_MAX_FWD, -1, 1)
                            kn.Position = UDim2.new(0.5, nx * maxR, 0.5, ny * maxR)
                        end
                        local skn = (side == "L") and ui.sliderLKnob or ui.sliderRKnob
                        if skn then
                            local ny = math.clamp(-oS.y / WHEEL_MAX_UP, -1, 1)
                            skn.Position = UDim2.new(0.5, 0, 0.5, ny * halfSlider)
                        end
                    end
                end
            end
        end

        local breath = math.sin(t * S.breathSpeed * math.pi * 2) * S.breathAmp
        local swayAmp = S.swayEnabled and S.handSway or 0
        local targetSwayL = Vector3.new(
            naturalSway(t, 1.31) * swayAmp,
            naturalSway(t, 2.73) * swayAmp * 0.7 + breath,
            naturalSway(t, 4.17) * swayAmp * 0.5)
        local targetSwayR = Vector3.new(
            naturalSway(t, 5.39) * swayAmp,
            naturalSway(t, 6.81) * swayAmp * 0.7 + breath,
            naturalSway(t, 8.23) * swayAmp * 0.5)
        local aS = 1 - math.exp(-6 * dt)
        swayOffsetL = swayOffsetL:Lerp(targetSwayL, aS)
        swayOffsetR = swayOffsetR:Lerp(targetSwayR, aS)

        if S.gyroEnabled and gyroBase and gyroNow then
            local rel = gyroBase:Inverse() * gyroNow
            local rx, ry, rz = rel:ToEulerAnglesYXZ()
            if math.abs(rx) > S.gyroAngleMax
            or math.abs(ry) > S.gyroAngleMax
            or math.abs(rz) > S.gyroAngleMax then
                gyroBase = gyroNow
                rx, ry, rz = 0, 0, 0
            end
            rx = math.clamp(rx, -S.gyroRotMax, S.gyroRotMax)
            ry = math.clamp(ry, -S.gyroRotMax, S.gyroRotMax)
            rz = math.clamp(rz, -S.gyroRotMax, S.gyroRotMax)

            local rL, pL, zL = rx, ry, rz
            local rR, pR, zR = rx, ry, rz
            if gyroSepEnabled then
                if selHand == "L" then rR, pR, zR = 0, 0, 0
                elseif selHand == "R" then rL, pL, zL = 0, 0, 0 end
            end

            local rk = S.gyroRotK
            local targetRL = CFrame.fromEulerAnglesYXZ(
                rL * rk * S.handMirrorL,
                pL * rk * S.handMirrorL,
                zL * rk * S.handMirrorL)
            local targetRR = CFrame.fromEulerAnglesYXZ(
                rR * rk * S.handMirrorR * -1,
                pR * rk * S.handMirrorR * -1,
                zR * rk * S.handMirrorR * -1)

            if S.swayEnabled then
                targetRL = targetRL * CFrame.fromEulerAnglesYXZ(
                    naturalSway(t, 10.1) * S.handRotSway,
                    naturalSway(t, 11.3) * S.handRotSway,
                    naturalSway(t, 12.7) * S.handRotSway)
                targetRR = targetRR * CFrame.fromEulerAnglesYXZ(
                    naturalSway(t, 13.1) * S.handRotSway,
                    naturalSway(t, 14.3) * S.handRotSway,
                    naturalSway(t, 15.7) * S.handRotSway)
            end

            local aR = 1 - math.exp(-S.gyroRotSmooth * dt)
            rotL = rotL:Lerp(targetRL, aR)
            rotR = rotR:Lerp(targetRR, aR)

            local pk = S.gyroPosK
            local targetGyroL = Vector3.new(
                -pL * pk * S.handMirrorL,
                zL * pk * S.handMirrorL * 0.6,
                -rL * pk * 0.4)
            local targetGyroR = Vector3.new(
                -pR * pk * S.handMirrorR,
                zR * pk * S.handMirrorR * 0.6,
                -rR * pk * 0.4)
            local maxP = S.gyroPosMax
            if targetGyroL.Magnitude > maxP then targetGyroL = targetGyroL.Unit * maxP end
            if targetGyroR.Magnitude > maxP then targetGyroR = targetGyroR.Unit * maxP end
            local aP = 1 - math.exp(-S.gyroRotSmooth * dt)
            gyroPosL = gyroPosL:Lerp(targetGyroL, aP)
            gyroPosR = gyroPosR:Lerp(targetGyroR, aP)

            if gyroSepEnabled and selHand ~= "B" then
                if selHand == "L" then
                    rotR = CFrame.new(); gyroPosR = Vector3.zero
                elseif selHand == "R" then
                    rotL = CFrame.new(); gyroPosL = Vector3.zero
                end
            end
        else
            local aR = math.min(dt * 6, 1)
            rotL = rotL:Lerp(CFrame.new(), aR); rotR = rotR:Lerp(CFrame.new(), aR)
            gyroPosL = gyroPosL:Lerp(Vector3.zero, aR); gyroPosR = gyroPosR:Lerp(Vector3.zero, aR)
        end

        local targetGPosL, targetGPosR = Vector3.zero, Vector3.zero
        local targetGRotL, targetGRotR = CFrame.new(), CFrame.new()
        if gestureCur then
            gestureT = gestureT + dt
            if gestureT >= gestureCur.dur then
                gestureCur = nil; gestureT = 0
            else
                local gp, gr = sampleGesture(gestureCur, gestureT)
                local h = gestureHand
                if h == "L" then
                    targetGPosL = Vector3.new(-gp.X, gp.Y, gp.Z); targetGRotL = gr
                elseif h == "R" then
                    targetGPosR = gp; targetGRotR = gr
                elseif h == "B" then
                    targetGPosL = Vector3.new(-gp.X, gp.Y, gp.Z); targetGRotL = gr
                    targetGPosR = gp; targetGRotR = gr
                end
            end
        end
        local aG = 1 - math.exp(-S.gestureBlend * dt)
        gesturePosL = gesturePosL:Lerp(targetGPosL, aG)
        gesturePosR = gesturePosR:Lerp(targetGPosR, aG)
        gestureRotL = gestureRotL:Lerp(targetGRotL, aG)
        gestureRotR = gestureRotR:Lerp(targetGRotR, aG)
        if spinMode then
            spinAngle = spinAngle + dt * 35
            local spinCF = CFrame.fromEulerAnglesXYZ(0, 0, spinAngle)
            rotL = rotL * spinCF; rotR = rotR * spinCF
        end
        local k = 1 - math.exp(-S.smooth * dt)
        local prevYaw = yaw
        local prevPitch = pitch
        yaw   = yaw   + (targetYaw   - yaw)   * k
        pitch = pitch + (targetPitch - pitch) * k
        if autoTiltEnabled then
            local yawVel = (yaw - prevYaw) / math.max(dt, 1/240)
            local autoRoll = math.clamp(yawVel * S.rollK, -S.rollMax, S.rollMax)
            local rollSpeed = (math.abs(autoRoll) > 0.005) and S.rollSmooth or S.rollReturnSmooth
            roll = roll + (autoRoll - roll) * (1 - math.exp(-rollSpeed * dt))
            local pitchVel = (pitch - prevPitch) / math.max(dt, 1/240)
            local autoPitchTilt = math.clamp(pitchVel * S.pitchTiltK, -S.pitchTiltMax, S.pitchTiltMax)
            local pitchSpeed = (math.abs(autoPitchTilt) > 0.005) and S.pitchTiltSmooth or S.pitchTiltReturnSmooth
            pitchTilt = pitchTilt + (autoPitchTilt - pitchTilt) * (1 - math.exp(-pitchSpeed * dt))
        else
            roll = roll * (1 - math.exp(-4 * dt))
            pitchTilt = pitchTilt * (1 - math.exp(-4 * dt))
        end
        local tiltK = 1 - math.exp(-9 * dt)
        manualRoll  = manualRoll  + (headTiltTargetRoll  - manualRoll)  * tiltK
        manualPitch = manualPitch + (headTiltTargetPitch - manualPitch) * tiltK
        local finalRoll = roll + manualRoll
        local finalPitchOffset = pitchTilt + manualPitch
        local headSwayYawV, headSwayPitchV = 0, 0
        local headSwayPosV = Vector3.zero
        if S.swayEnabled then
            headSwayYawV   = (slowDrift(t, 20.1) * 0.7 + naturalSway(t, 20.1) * 0.3) * S.headSwayYaw
            headSwayPitchV = (slowDrift(t, 33.7) * 0.7 + naturalSway(t, 33.7) * 0.3) * S.headSwayPitch
            headSwayPosV   = Vector3.new(
                slowDrift(t, 22.7) * S.headSwayPos,
                slowDrift(t, 23.9) * S.headSwayPos * 1.2 + breathDrift(t, 40.3) * S.headSwayPos * 0.8,
                slowDrift(t, 25.1) * S.headSwayPos * 0.5)
        end
        local breathHead = breath * S.breathHeadK
        headSwayPosV = headSwayPosV + Vector3.new(0, breathHead, 0)
        headSwayPitchV = headSwayPitchV + breathHead * 0.4
        local finalPitch = pitch + headSwayPitchV + finalPitchOffset
        local finalYaw = yaw + headSwayYawV
        local rot = CFrame.fromEulerAnglesYXZ(finalPitch, finalYaw, finalRoll)
        local rotDelta = prevRotForInertia:Inverse() * rot
        local idy, idp, _ = rotDelta:ToEulerAnglesYXZ()
        handInertiaVec = handInertiaVec + Vector3.new(idy * S.handInertia, -idp * S.handInertia, 0)
        handInertiaVec = handInertiaVec * math.exp(-S.handInertiaDecay * dt)
        if handInertiaVec.Magnitude > 0.35 then handInertiaVec = handInertiaVec.Unit * 0.35 end
        prevRotForInertia = rot
        local hs = cam.HeadScale
        if type(hs) ~= "number" or hs <= 1 then hs = S.scale * 6 end
        local spd = (10 + S.scale * 4) * hs * S.moveK * S.speedMult
        local mv = Vector3.zero
        if keys[Enum.KeyCode.W] then mv += Vector3.new(0,0,-1) end
        if keys[Enum.KeyCode.S] then mv += Vector3.new(0,0, 1) end
        if keys[Enum.KeyCode.A] then mv += Vector3.new(-1,0,0) end
        if keys[Enum.KeyCode.D] then mv += Vector3.new( 1,0,0) end
        if keys[Enum.KeyCode.Space] then mv += Vector3.new(0, 1,0) end
        if keys[Enum.KeyCode.LeftShift] then mv += Vector3.new(0,-1,0) end
        local targetVel = Vector3.zero
        if mv.Magnitude > 0 then targetVel = (rot * mv.Unit) * spd end
        moveVel = targetVel
        camPos = camPos + moveVel * dt
        _G.__NoVR_camPos = camPos
        cam.CameraType = Enum.CameraType.Scriptable
        cam.CFrame = CFrame.new(camPos + headSwayPosV) * rot

        if freeCamMode and freeCamBodyCFrame and _G.__NoVR_rig then
            local vrHead = _G.__NoVR_rig:FindFirstChild("VRHead")
            if vrHead then
                pcall(function()
                    vrHead.CFrame = freeCamBodyCFrame
                end)
            end
        end

        if Input then
            Input.directionLateral = Vector2.zero
            Input.directionVertical = 0
            Input.turnDirection = 0
        end

        if recStatusRef then
            if recordMode then
                recStatusRef.Visible = true
                recStatusRef.BackgroundColor3 = Color3.fromRGB(200, 30, 30)
                recStatusRef.Text = ("● 录制 %.1fs"):format(t - recordStartT)
            elseif playbackMode then
                recStatusRef.Visible = true
                recStatusRef.BackgroundColor3 = Color3.fromRGB(30, 100, 200)
                recStatusRef.Text = ("▶ 回放 %.1f/%.1f"):format(playbackFrame / 60, maxRecordFrame / 60)
            elseif freeCamMode then
                recStatusRef.Visible = true
                recStatusRef.BackgroundColor3 = Color3.fromRGB(120, 80, 200)
                recStatusRef.Text = "👁 自由头部视角"
            else
                recStatusRef.Visible = false
            end
        end
    end)

        local function collectThemeTargets()
            themeTargetsMain = {}
            for _, obj in ipairs(mobileGui:GetDescendants()) do
                local insideTheme = false
                local p = obj.Parent
                while p do
                    if p == themePanel then insideTheme = true; break end
                    p = p.Parent
                end
                if not insideTheme then
                    if obj:IsA("TextButton") or obj:IsA("TextLabel") or obj:IsA("Frame") or obj:IsA("TextBox") then
                        table.insert(themeTargetsMain, obj)
                    end
                end
            end

            themeTargetsWheel = {}
            local wg = _G.__NoVR_wheelGui
            if wg then
                for _, obj in ipairs(wg:GetDescendants()) do
                    if obj.Name == "WheelBg" or obj.Name == "SliderBg"
                       or obj:IsA("TextLabel") then
                        table.insert(themeTargetsWheel, obj)
                    end
                end
            end
        end

        local function applyTheme()
            local ap = themePanel.AbsolutePosition
            local as = themePanel.AbsoluteSize
            local cx, cy = ap.X + as.X * 0.5, ap.Y + as.Y * 0.5
            local kp = themeKnob.AbsolutePosition
            local ks = themeKnob.AbsoluteSize
            local kx, ky = kp.X + ks.X * 0.5, kp.Y + ks.Y * 0.5
            local dx, dy = kx - cx, ky - cy
            local r = math.sqrt(dx * dx + dy * dy)
            local maxR = as.X * 0.5
            local sat = math.clamp(r / maxR, 0, 1)
            local angle = math.atan2(dy, dx)
            local hue = (angle / (math.pi * 2)) % 1
            if hue < 0 then hue = hue + 1 end

            local bgColor     = Color3.fromHSV(hue, sat * 0.55, 0.16)
            local strokeColor = Color3.fromHSV(hue, sat, 0.95)
            local textColor   = Color3.fromHSV(hue, math.min(sat * 0.25, 0.25), 1)

            themeKnob.BackgroundColor3 = Color3.fromHSV(hue, sat, 1)

            if not themeTargetsMain then collectThemeTargets() end

            for _, obj in ipairs(themeTargetsMain) do
                if obj:IsA("TextButton") or obj:IsA("TextLabel") then
                    if obj.BackgroundTransparency < 0.99 then
                        obj.BackgroundColor3 = bgColor
                    end
                    obj.TextColor3 = textColor
                    local s = obj:FindFirstChildOfClass("UIStroke")
                    if s then s.Color = strokeColor end
                elseif obj:IsA("Frame") or obj:IsA("TextBox") then
                    if obj.BackgroundTransparency < 0.99 then
                        obj.BackgroundColor3 = bgColor
                    end
                    if obj:IsA("TextBox") then obj.TextColor3 = textColor end
                end
            end

            for _, obj in ipairs(themeTargetsWheel) do
                if obj.Name == "WheelBg" or obj.Name == "SliderBg" then
                    obj.BackgroundColor3 = bgColor
                    local s = obj:FindFirstChildOfClass("UIStroke")
                    if s then s.Color = strokeColor end
                elseif obj:IsA("TextLabel") then
                    obj.TextColor3 = textColor
                end
            end
        end

        local themeDirty = false
        local themeScheduled = false
        local function scheduleApplyTheme()
            themeDirty = true
            if themeScheduled then return end
            themeScheduled = true
            task.spawn(function()
                while themeDirty do
                    themeDirty = false
                    applyTheme()
                    task.wait()
                end
                themeScheduled = false
            end)
        end

            local maxR = as.X * 0.5 - 16

            local dy = pos.Y - cy

            themeKnob.Position = UDim2.new(0.5, dx, 0.5, dy)

        end
